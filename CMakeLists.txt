cmake_minimum_required(VERSION 3.20)
project(KBL VERSION 0.3.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(kbl_core src/core.cpp src/hardware_menu.cpp src/boot_image.cpp src/fdt.cpp src/linux_boot.cpp src/buttons.cpp src/menu_view.cpp src/font.cpp)
target_include_directories(kbl_core PUBLIC include)
if(MSVC)
  target_compile_options(kbl_core PRIVATE /W4 /WX)
else()
  target_compile_options(kbl_core PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_executable(kbl platforms/host/main.cpp)
target_link_libraries(kbl PRIVATE kbl_core)
add_library(kbl_gta9wifi targets/samsung/gta9wifi/display_probe.cpp targets/samsung/gta9wifi/controls.cpp)
target_include_directories(kbl_gta9wifi PUBLIC targets/samsung/gta9wifi)
target_link_libraries(kbl_gta9wifi PUBLIC kbl_core)
target_link_libraries(kbl PRIVATE kbl_gta9wifi)
include(CTest)
if(BUILD_TESTING)
  add_executable(kbl_tests tests/core_tests.cpp)
  target_link_libraries(kbl_tests PRIVATE kbl_core)
  add_test(NAME kbl_core COMMAND kbl_tests)
  add_executable(kbl_hardware_menu_tests tests/hardware_menu_tests.cpp)
  target_link_libraries(kbl_hardware_menu_tests PRIVATE kbl_core)
  add_test(NAME kbl_hardware_menu COMMAND kbl_hardware_menu_tests)
  add_executable(kbl_display_tests tests/display_tests.cpp)
  target_link_libraries(kbl_display_tests PRIVATE kbl_gta9wifi)
  add_test(NAME kbl_display COMMAND kbl_display_tests)
  add_executable(kbl_controls_tests tests/controls_tests.cpp)
  target_link_libraries(kbl_controls_tests PRIVATE kbl_gta9wifi)
  add_test(NAME kbl_controls COMMAND kbl_controls_tests)
  add_executable(kbl_linux_tests tests/linux_boot_tests.cpp)
  target_link_libraries(kbl_linux_tests PRIVATE kbl_gta9wifi)
  add_test(NAME kbl_linux COMMAND kbl_linux_tests)
endif()
